/* ================= INITIALIZATION & LOCAL STORAGE ================= */
function getUserData() {
const data = localStorage.getItem('smokefree_user_data');
if (data) {
return JSON.parse(data);
}
// Defaults to 1 day ago for demonstration if brand new
const defaultDate = new Date();
defaultDate.setHours(defaultDate.getHours() - 12);

const defaultSettings = {
quitDate: defaultDate.toISOString(),
cigsPerDay: 10,
costPerCig: 15,
currency: '₹'
};
localStorage.setItem('smokefree_user_data', JSON.stringify(defaultSettings));
return defaultSettings;
}

function getCravingLogs() {
const logs = localStorage.getItem('smokefree_craving_logs');
return logs ? JSON.parse(logs) : [];
}

/* ================= UI RENDERERS ================= */
function updateDashboard() {
const user = getUserData();
const quitTime = new Date(user.quitDate).getTime();
const now = new Date().getTime();
const diffMs = Math.max(0, now - quitTime);

const totalSeconds = Math.floor(diffMs / 1000);
const days = Math.floor(totalSeconds / 86400);
const hours = Math.floor((totalSeconds % 86400) / 3600);
const minutes = Math.floor((totalSeconds % 3600) / 60);
const seconds = totalSeconds % 60;

// Update Live Timers
document.getElementById('streak-days').innerText = days;
document.getElementById('streak-hours').innerText = hours.toString().padStart(2, '0');
document.getElementById('streak-minutes').innerText = minutes.toString().padStart(2, '0');
document.getElementById('streak-seconds').innerText = seconds.toString().padStart(2, '0');

// Calculations
const daysPassedExact = totalSeconds / 86400;
const cigsAvoided = Math.floor(daysPassedExact * user.cigsPerDay);
const moneySaved = Math.round(cigsAvoided * user.costPerCig);
const lifeRegainedMinutes = cigsAvoided * 11; // ~11 minutes per cigarette

document.getElementById('stat-money').innerText = `₹${moneySaved.toLocaleString()}`;
document.getElementById('stat-cigs').innerText = `${cigsAvoided} টি`;

if (lifeRegainedMinutes >= 60) {
const hrs = Math.floor(lifeRegainedMinutes / 60);
const mins = lifeRegainedMinutes % 60;
document.getElementById('stat-life').innerText = `${hrs}ঘ ${mins}মি`;
} else {
document.getElementById('stat-life').innerText = `${lifeRegainedMinutes} মি`;
}

const qDate = new Date(user.quitDate);
document.getElementById('quit-start-badge').innerText = `শুরু: ${qDate.toLocaleDateString('bn-BD', { day:'numeric', month:'short' })}`;

// Resisted Cravings count
const cravings = getCravingLogs();
document.getElementById('stat-resisted').innerText = `${cravings.length} বার`;

// Update Health Timeline progress bars
renderHealthTimeline(totalSeconds);
}

function renderHealthTimeline(totalSecondsPassed) {
const container = document.getElementById('health-timeline-container');
if (!container) return;

container.innerHTML = HEALTH_MILESTONES.map(item => {
const percent = Math.min(100, Math.round((totalSecondsPassed / item.durationSec) * 100));
const isComplete = percent >= 100;

return `
<div class="bg-cardBg rounded-2xl p-4 border border-slate-800 transition">
<div class="flex justify-between items-start mb-1.5">
<div>
<span class="text-xs font-bold ${isComplete ? 'text-emerald-400' : 'text-slate-200'}">${item.title}</span>
<p class="text-xs text-slate-400 mt-0.5 leading-snug">${item.desc}</p>
</div>
<div class="text-right ml-2 shrink-0">
<span class="text-xs font-bold ${isComplete ? 'text-emerald-400' : 'text-slate-400'}">${percent}%</span>
</div>
</div>
<!-- Progress Bar -->
<div class="w-full bg-slate-900 rounded-full h-2 overflow-hidden mt-2">
<div class="h-2 rounded-full transition-all duration-500 ${isComplete ? 'bg-gradient-to-r from-emerald-500 to-teal-400' : 'bg-emerald-500/70'}" style="width: ${percent}%"></div>
</div>
</div>
`;
}).join('');
}

function switchTab(tab) {
const tabs = ['dashboard', 'health', 'cravings'];
tabs.forEach(t => {
const el = document.getElementById(`tab-${t}`);
const navEl = document.getElementById(`nav-${t}`);
if (t === tab) {
el.classList.remove('hidden');
navEl.classList.remove('text-slate-400');
navEl.classList.add('text-emerald-400', 'font-medium');
} else {
el.classList.add('hidden');
navEl.classList.remove('text-emerald-400', 'font-medium');
navEl.classList.add('text-slate-400');
}
});
if (tab === 'cravings') {
renderCravingLogs();
}
}

function openSettingsModal() {
const user = getUserData();
const dateObj = new Date(user.quitDate);
// Format to YYYY-MM-DDTHH:MM for input
const localIso = new Date(dateObj.getTime() - dateObj.getTimezoneOffset() * 60000).toISOString().slice(0, 16);

document.getElementById('input-quit-date').value = localIso;
document.getElementById('input-cigs-day').value = user.cigsPerDay;
document.getElementById('input-cig-cost').value = user.costPerCig;
document.getElementById('settings-modal').classList.remove('hidden');
}

function closeSettingsModal() {
document.getElementById('settings-modal').classList.add('hidden');
}

function saveSettings(e) {
e.preventDefault();
const quitDate = new Date(document.getElementById('input-quit-date').value).toISOString();
const cigsPerDay = parseInt(document.getElementById('input-cigs-day').value) || 10;
const costPerCig = parseFloat(document.getElementById('input-cig-cost').value) || 15;

const updated = {
quitDate,
cigsPerDay,
costPerCig,
currency: '₹'
};

localStorage.setItem('smokefree_user_data', JSON.stringify(updated));
closeSettingsModal();
showToast("সেটিংস সফলভাবে সংরক্ষিত হয়েছে!");
updateDashboard();
}

function selectTrigger(triggerName) {
currentTrigger = triggerName;
const buttons = document.querySelectorAll('.trigger-btn');
buttons.forEach(btn => {
if (btn.innerText.includes(triggerName.split(' ')[0])) {
btn.classList.add('border-emerald-500/40', 'text-slate-200');
btn.classList.remove('border-slate-700', 'text-slate-300');
} else {
btn.classList.remove('border-emerald-500/40', 'text-slate-200');
btn.classList.add('border-slate-700', 'text-slate-300');
}
});
}

function saveCravingLog() {
const noteInput = document.getElementById('craving-note');
const note = noteInput.value.trim();

const newEntry = {
id: Date.now(),
trigger: currentTrigger,
note: note,
timestamp: new Date().toISOString()
};

const logs = getCravingLogs();
logs.unshift(newEntry);
localStorage.setItem('smokefree_craving_logs', JSON.stringify(logs));

noteInput.value = '';
showToast("অভিনন্দন! আপনি প্রলোভন জয় করেছেন।");
renderCravingLogs();
updateDashboard();
}

function toggleBreathingGuide() {
if (isBreathingActive) {
stopBreathingGuide();
} else {
startBreathingGuide();
}
}

function startBreathingGuide() {
isBreathingActive = true;
document.getElementById('breathe-toggle-btn').innerText = "থামুন";
document.getElementById('breathe-toggle-btn').classList.replace('bg-emerald-500', 'bg-slate-700');
document.getElementById('breathe-toggle-btn').classList.replace('text-slate-900', 'text-white');

const circle = document.getElementById('breathe-circle');
const phaseText = document.getElementById('breathe-phase');
const countText = document.getElementById('breathe-countdown');

let currentStep = 'in'; // 'in', 'hold', 'out'
let timeLeft = 4;

function step() {
if (!isBreathingActive) return;

if (currentStep === 'in') {
phaseText.innerText = "শ্বাস নিন";
circle.style.transform = 'scale(1.35)';
countText.innerText = `${timeLeft} সেকেন্ড`;
timeLeft--;
if (timeLeft < 0) {
currentStep = 'hold';
timeLeft = 7;
}
} else if (currentStep === 'hold') {
phaseText.innerText = "ধরে রাখুন";
countText.innerText = `${timeLeft} সেকেন্ড`;
timeLeft--;
if (timeLeft < 0) {
currentStep = 'out';
timeLeft = 8;
}
} else if (currentStep === 'out') {
phaseText.innerText = "ধীরে ছাড়ুন";
circle.style.transform = 'scale(1.0)';
countText.innerText = `${timeLeft} সেকেন্ড`;
timeLeft--;
if (timeLeft < 0) {
currentStep = 'in';
timeLeft = 4;
}
}
}

step();
breathingInterval = setInterval(step, 1000);
}

function stopBreathingGuide() {
isBreathingActive = false;
clearInterval(breathingInterval);
const circle = document.getElementById('breathe-circle');
circle.style.transform = 'scale(1.0)';
document.getElementById('breathe-phase').innerText = "প্রস্তুত";
document.getElementById('breathe-countdown').innerText = "শুরু করুন";
const btn = document.getElementById('breathe-toggle-btn');
btn.innerText = "অনুশীলন শুরু করুন";
btn.classList.replace('bg-slate-700', 'bg-emerald-500');
btn.classList.replace('text-white', 'text-slate-900');
}

function showToast(message) {
const toast = document.getElementById('toast');
document.getElementById('toast-msg').innerText = message;
toast.classList.remove('opacity-0');
toast.classList.add('opacity-100');
setTimeout(() => {
toast.classList.remove('opacity-100');
toast.classList.add('opacity-0');
}, 2500);
}
